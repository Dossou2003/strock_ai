Fichier: chatbot/gemini_client.py

import google.generativeai as genai

class GeminiChatbot:
    def __init__(self):
        genai.configure(api_key=settings.GEMINI_API_KEY)
        self.model = genai.GenerativeModel('gemini-pro')
        
    def get_response(self, user_message, context=None):
        """
        Génère une réponse avec contexte médical
        
        Args:
            user_message: Question utilisateur
            context: Contexte (résultats analyse, etc.)
        """
        system_prompt = """
        Tu es un assistant médical spécialisé en AVC.
        Tu expliques les résultats d'analyse de façon claire et rassurante.
        Tu ne poses pas de diagnostic mais tu éduques.
        """
        
        full_prompt = f"{system_prompt}\n\nContexte: {context}\n\nQuestion: {user_message}"
        response = self.model.generate_content(full_prompt)
        return response.text

  )

INTÉGRATION DANS LE CODE:
--------------------------

backend/ai_models/unet_model.py:
```python
from tensorflow.keras.models import load_model
import os

class UNetSegmenter:
    def __init__(self):
        model_path = os.path.join(settings.BASE_DIR, 'models_saved', 'unet_territories.h5')
        self.model = load_model(model_path)
    
    def predict(self, image):
        # Prétraitement
        img_resized = cv2.resize(image, (256, 256)) / 255.0
        img_input = img_resized[None, ..., None]
        
        # Prédiction
        pred = self.model.predict(img_input)[0]
        pred_mask = np.argmax(pred, axis=-1)
        
        return pred_mask
```

backend/ai_models/svm_classifier.py:
```python
import joblib

class StrokeClassifier:
    def __init__(self):
        self.model = joblib.load('models_saved/svm_classifier.pkl')
        self.scaler = joblib.load('models_saved/scaler.pkl')
    
    def predict(self, features_vector):
        # Normalisation
        features_scaled = self.scaler.transform([features_vector])
        
        # Prédiction
        prediction = self.model.predict(features_scaled)[0]
        probability = self.model.predict_proba(features_scaled)[0][1]
        
        return {
            'has_stroke': bool(prediction),
            'probability': float(probability)
        }
```
